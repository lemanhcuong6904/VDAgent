LangChain agent + middleware · long-term memory

    
    
Before the loop it finds this user's earlier findings closest to the request and puts them in the system prompt. After the answer it decides what is worth remembering and skips what it already knows.

    
class Memory(AgentMiddleware):
    async def abefore_agent(self, state, runtime):
        v = await embeddings.aembed_query(request)
        notes = await ctx.memory.search(request, 5, embedding=v)
        self.recall = render(notes)          # added to every system prompt

    async def aafter_agent(self, state, runtime):
        for finding in await extract(request, answer):   # at most 3
            v = await embeddings.aembed_query(finding)
            nearest = await ctx.memory.search(finding, 1, embedding=v)
            if not nearest or nearest[0].score >= 0.1:   # not already known
                await ctx.memory.save(finding, "finding", v)

create_agent(model, tools, middleware=[Memory(ctx), CtxBridge(ctx)])
